cmake_minimum_required(VERSION 3.20)
project(sunburn_sim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "Godot API")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "Bindings")
add_subdirectory(vendor/godot-cpp)
add_library(sunburn_sim SHARED src/simulation.cpp)
target_link_libraries(sunburn_sim PRIVATE godot::cpp)
target_compile_options(sunburn_sim PRIVATE -O3 -fvisibility=hidden -fno-exceptions)
set_target_properties(sunburn_sim PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
if(EMSCRIPTEN)
 set_target_properties(sunburn_sim PROPERTIES SUFFIX ".wasm" PREFIX "")
 target_link_options(sunburn_sim PRIVATE -sSIDE_MODULE=1 -sSUPPORT_LONGJMP=wasm)
endif()
